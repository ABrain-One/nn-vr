#!/usr/bin/env python3
"""Compare retained NN-Dataset PyTorch records with canonical A/B summaries.

This script is intentionally read-only with respect to NN-Dataset and raw A/B
campaign artifacts.  It reads their retained JSON/CSV records and writes only
the derived files in this directory.
"""

from __future__ import annotations

import csv
import json
import math
import os
from collections import Counter
from pathlib import Path
from statistics import median
from typing import Any

os.environ.setdefault("MPLCONFIGDIR", "/tmp/nnvr-matplotlib")

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
from scipy.stats import spearmanr


ROOT = Path(__file__).resolve().parents[3]
OUT = Path(__file__).resolve().parent
PT_ROOT = ROOT / "nn-dataset" / "ab" / "nn" / "stat" / "run" / "pt" / "fp32"
STRUCTURED_PATH = (
    ROOT
    / "nn-dataset"
    / "ab"
    / "nn"
    / "stat"
    / "run"
    / "pt"
    / "structured_l1_layrewise"
    / "img-classification_cifar-10_ac"
    / "all_models.json"
)
AB_PER_MODEL = ROOT / "analysis" / "experiment_ab" / "per_model_comparison.csv"
A_V1_ROOT = ROOT / "out" / "nn" / "stat" / "barracuda_completion_aware_v1" / "onnx" / "fp32"

PT_HOST_FILE = "PT_HOST.json"
A_HOST_FILE = "UNITY_HOST.json"
PT_DIR_PREFIX = "img-classification_cifar-10_acc_"


def relative(path: Path) -> str:
    return path.relative_to(ROOT).as_posix()


def as_bool(value: str | bool | None) -> bool:
    return str(value).strip().lower() == "true"


def float_or_none(value: Any) -> float | None:
    try:
        result = float(value)
    except (TypeError, ValueError):
        return None
    return result if math.isfinite(result) else None


def shape_string(shape: tuple[Any, ...] | None) -> str:
    return "" if not shape else "x".join(str(x) for x in shape)


def quantile(values: list[float], q: float) -> float:
    return float(np.quantile(np.asarray(values, dtype=float), q))


def write_csv(path: Path, rows: list[dict[str, Any]]) -> None:
    if not rows:
        raise RuntimeError(f"Refusing to write empty derived table: {path}")
    fields = list(rows[0])
    with path.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)


def read_json(path: Path) -> dict[str, Any]:
    return json.loads(path.read_text(encoding="utf-8"))


def source_catalog() -> tuple[dict[str, tuple[Path, dict[str, Any]]], dict[str, Any]]:
    """Load the relevant CIFAR-10 PT corpus and reject duplicate identities."""
    files = sorted(PT_ROOT.glob(f"{PT_DIR_PREFIX}*/{PT_HOST_FILE}"))
    catalog: dict[str, tuple[Path, dict[str, Any]]] = {}
    duplicate_names: list[str] = []
    valid_count = 0
    iterations = Counter()
    units = Counter()
    devices = Counter()
    os_versions = Counter()
    cpu_cores = Counter()
    shapes = Counter()
    for path in files:
        record = read_json(path)
        model = record.get("model_name")
        if not isinstance(model, str) or not model:
            raise RuntimeError(f"Missing model_name in {path}")
        if model in catalog:
            duplicate_names.append(model)
            continue
        catalog[model] = (path, record)
        if record.get("valid") is True:
            valid_count += 1
        iterations[record.get("iterations")] += 1
        units[record.get("unit")] += 1
        devices[record.get("device_type")] += 1
        os_versions[record.get("os_version")] += 1
        cpu_info = record.get("device_analytics", {}).get("cpu_info", {})
        cpu_cores[cpu_info.get("cpu_cores")] += 1
        shapes[tuple(record.get(f"in_dim_{i}") for i in range(4))] += 1
    if duplicate_names:
        raise RuntimeError(f"Duplicate PT model identities: {sorted(set(duplicate_names))[:10]}")
    return catalog, {
        "all_pt_fp32_records_across_tasks": len(list(PT_ROOT.glob("*/*.json"))),
        "cifar10_records_on_selected_host": len(files),
        "cifar10_unique_model_names": len(catalog),
        "cifar10_valid_records": valid_count,
        "iterations": dict(sorted(iterations.items(), key=lambda item: str(item[0]))),
        "gpu_units": dict(sorted(units.items())),
        "device_types": dict(sorted(devices.items())),
        "os_versions": dict(sorted(os_versions.items())),
        "logical_cpu_counts": dict(sorted(cpu_cores.items(), key=lambda item: str(item[0]))),
        "input_shapes_nhwc": {shape_string(shape): count for shape, count in sorted(shapes.items())},
    }


def a_input_shape(model: str) -> tuple[tuple[Any, ...] | None, str | None]:
    path = A_V1_ROOT / f"{PT_DIR_PREFIX}{model}" / A_HOST_FILE
    if not path.is_file():
        return None, None
    record = read_json(path)
    value = record.get("input_shape")
    return (tuple(value) if isinstance(value, list) else None), relative(path)


def latency_summary(rows: list[dict[str, Any]], backend: str, unity: str) -> dict[str, Any]:
    pt = [float(row[f"pt_{backend}_ms"]) for row in rows]
    runtime = [float(row[f"{unity}_{backend}_ms"]) for row in rows]
    differences = [u - p for p, u in zip(pt, runtime)]
    ratios = [u / p for p, u in zip(pt, runtime)]
    result = spearmanr(pt, runtime)
    return {
        "n": len(rows),
        "pt_median_ms": median(pt),
        f"{unity}_median_ms": median(runtime),
        "pt_iqr_ms": [quantile(pt, 0.25), quantile(pt, 0.75)],
        f"{unity}_iqr_ms": [quantile(runtime, 0.25), quantile(runtime, 0.75)],
        f"median_{unity}_minus_pt_ms": median(differences),
        f"median_{unity}_over_pt": median(ratios),
        f"{unity}_over_pt_ratio_quantiles": [quantile(ratios, q) for q in (0.05, 0.25, 0.5, 0.75, 0.95)],
        f"{unity}_faster_than_pt_count": sum(u < p for p, u in zip(pt, runtime)),
        f"{unity}_faster_than_pt_percent": 100.0 * sum(u < p for p, u in zip(pt, runtime)) / len(rows),
        f"spearman_pt_vs_{unity}_rho": float(result.statistic),
        f"spearman_pt_vs_{unity}_p_value": float(result.pvalue),
    }


def closeness_summary(rows: list[dict[str, Any]], backend: str) -> dict[str, Any]:
    a = [float(row[f"a_{backend}_over_pt"]) for row in rows]
    b = [float(row[f"b_{backend}_over_pt"]) for row in rows]
    a_distance = [abs(value - 1.0) for value in a]
    b_distance = [abs(value - 1.0) for value in b]
    return {
        "definition": "smaller absolute per-model ratio distance |Unity/PT - 1|",
        "b_closer_count": sum(bv < av for av, bv in zip(a_distance, b_distance)),
        "a_closer_count": sum(av < bv for av, bv in zip(a_distance, b_distance)),
        "tie_count": sum(av == bv for av, bv in zip(a_distance, b_distance)),
        "median_abs_ratio_distance_a": median(a_distance),
        "median_abs_ratio_distance_b": median(b_distance),
    }


def make_plot(rows: list[dict[str, Any]]) -> None:
    colors = {"a": "#cf772a", "b": "#337a58"}
    labels = {"a": "Barracuda A", "b": "Inference Engine B"}
    fig, axes = plt.subplots(2, 1, figsize=(6.4, 7.0), constrained_layout=True)
    for axis, backend in zip(axes, ("cpu", "gpu")):
        values = []
        for runtime in ("a", "b"):
            x = np.array([float(row[f"pt_{backend}_ms"]) for row in rows])
            y = np.array([float(row[f"{runtime}_{backend}_ms"]) for row in rows])
            values.extend(x.tolist() + y.tolist())
            axis.scatter(x, y, s=10, alpha=0.40, color=colors[runtime], edgecolors="none", label=labels[runtime])
        lower = min(values) * 0.8
        upper = max(values) * 1.2
        axis.plot([lower, upper], [lower, upper], "--", color="black", linewidth=0.9, label="1:1")
        axis.set_xscale("log")
        axis.set_yscale("log")
        axis.set_xlim(lower, upper)
        axis.set_ylim(lower, upper)
        axis.set_xlabel(f"Standalone PyTorch FP32 {backend.upper()} (ms)")
        axis.set_ylabel(f"Unity completion-aware {backend.upper()} (ms)")
        axis.set_title(f"{backend.upper()} timing, identity-matched N={len(rows)}")
        axis.grid(True, which="both", linewidth=0.3, alpha=0.35)
        axis.legend(loc="upper left", fontsize=8)
    fig.suptitle("Same-host retained records; log axes; 1:1 is descriptive only", fontsize=10)
    fig.savefig(OUT / "pt_unity_scatter.pdf", bbox_inches="tight")
    fig.savefig(OUT / "pt_unity_scatter.png", dpi=220, bbox_inches="tight")
    plt.close(fig)


def markdown_number(value: float, digits: int = 4) -> str:
    return f"{value:.{digits}f}"


def make_source_selection_document(summary: dict[str, Any]) -> str:
    catalog = summary["pt_catalog"]
    structured = summary["structured_l1_layerwise"]
    return f"""# NN-Dataset source selection

## Decision

Use `nn-dataset/ab/nn/stat/run/pt/fp32/`, restricted to
`img-classification_cifar-10_acc_*/PT_HOST.json`, as the retained
standalone PyTorch source. The relevant subset contains
{catalog['cifar10_records_on_selected_host']:,} exact model records on the
selected host, of which {catalog['cifar10_valid_records']:,} are marked valid.
The wider `pt/fp32` root contains {catalog['all_pt_fp32_records_across_tasks']:,}
records across tasks/datasets and is not itself the comparison population.

The selected JSON schema stores an exact `model_name`, validity/error fields,
the `Linux_MS-7E27` host identity, model-specific input dimensions, 20
iterations, CPU/GPU mean/min/max/standard-deviation durations, and device
analytics. Durations are nanoseconds and are converted to milliseconds only in
the derived comparison files.

## Why this is the correct source

`nn-dataset/ab/nn/util/Inference.py` is the retained producer for these files.
It constructs each selected model, calls `eval()`, uses 10 warm-ups and 20
timed iterations, records CPU timings with `time.perf_counter()`, and records
GPU timings with CUDA events followed by `torch.cuda.synchronize()`. The
existing historical manuscript also identifies `pt/fp32` as its standalone
PyTorch/workstation source.

The source records show `{catalog['gpu_units']}` for the GPU unit and
`{catalog['device_types']}` for the board identity. Together with the retained
A/B host records, these identify the same physical MS-7E27 / RTX 4090 host.
The PT JSON CPU metadata records 16 logical cores but only the generic
`x86_64` processor string; the more specific Ryzen 7 7800X3D identity is
retained in the A/B environment records.

## Why `structured_l1_layrewise` is rejected

`nn-dataset/ab/nn/stat/run/pt/structured_l1_layrewise/`
`img-classification_cifar-10_ac/all_models.json` is one aggregate
structured-L1 layerwise-pruning summary. It has {structured['model_count']}
model entries ({structured['overlap_with_original_manifest']} exact names from
the 1,762-model manifest; {structured['status_counts']}) and records pruning ratio,
post-pruning accuracy, parameter/model-size changes, and one `duration` field.
It has no per-model standalone CPU/GPU timing pair, device metadata, input
shape, warm-up/iteration provenance, or completion-boundary information.
It is therefore a pruning/accuracy analysis, not a baseline inference-latency
source, and is not used in the PT/A/B comparison.

## Matching rule

Matching is exact `model_name` equality only. No aliases, prefixes, fuzzy
matching, or checkpoint-name substitutions are applied. The eight original
manifest entries lacking a selected PT record are listed in
`comparison_summary.json` and `model_overlap.csv`.

## Comparability conclusion

The retained source supports **Level 3: context/rank comparison only**. It
shares host, model identifiers, batch size one, FP32 source designation, and
20 measurement iterations with A/B, but not the final timing boundary:

- PT CPU measures synchronous PyTorch forward-return time without an explicit
  output-array materialization.
- PT GPU uses CUDA event elapsed time and synchronizes after the end event, but
  does not include host output readback.
- A/B time host-output materialization in every warm-up and measured iteration.
- The PT runner instantiates `Net` directly and does not load the epoch-50
  checkpoint used by the Unity exports.
- Only {summary['shape_audit']['shape_aligned_count']:,}/
  {summary['overlap']['pt_a_b_identity_matched']} identity-matched models have
  the same retained PT and A input shapes.

Accordingly, paired ratios and rank correlations are descriptive context, not
portable Unity-overhead or speedup estimates.
"""

def main() -> None:
    ab_rows = list(csv.DictReader(AB_PER_MODEL.open(encoding="utf-8", newline="")))
    if len(ab_rows) != 1762:
        raise RuntimeError(f"Expected canonical 1,762-row A/B population, got {len(ab_rows)}")
    if len({row['model_name'] for row in ab_rows}) != len(ab_rows):
        raise RuntimeError("Duplicate model names in canonical A/B per-model CSV")

    pt_catalog, catalog_summary = source_catalog()
    structured = read_json(STRUCTURED_PATH)
    structured_statuses = Counter(
        entry.get("status", "missing") for entry in structured.values() if isinstance(entry, dict)
    )
    ab_by_model = {row["model_name"]: row for row in ab_rows}

    overlap_rows: list[dict[str, Any]] = []
    matched_rows: list[dict[str, Any]] = []
    missing_pt: list[str] = []
    shape_pair_counts = Counter()
    for ab in ab_rows:
        model = ab["model_name"]
        a_pass = as_bool(ab["a_repeat_stable_pass"])
        b_pass = as_bool(ab["b_repeat_stable_pass"])
        pt_entry = pt_catalog.get(model)
        pt_path: Path | None = None
        pt_record: dict[str, Any] | None = None
        if pt_entry is not None:
            pt_path, pt_record = pt_entry
        pt_valid = bool(pt_record and pt_record.get("valid") is True)
        a_shape, a_path = a_input_shape(model)
        pt_shape = tuple(pt_record.get(f"in_dim_{i}") for i in range(4)) if pt_record else None
        shape_match = bool(pt_shape and a_shape and pt_shape == a_shape)
        if pt_shape and a_shape:
            shape_pair_counts[(shape_string(pt_shape), shape_string(a_shape))] += 1
        if not pt_record:
            status = "missing_selected_pt_record"
            missing_pt.append(model)
        elif not pt_valid:
            status = "selected_pt_record_not_valid"
        elif not a_pass or not b_pass:
            status = "not_in_a_b_shared_strict_success"
        elif shape_match:
            status = "pt_a_b_identity_matched_shape_aligned"
        else:
            status = "pt_a_b_identity_matched_shape_mismatch"
        overlap_rows.append(
            {
                "model_name": model,
                "in_original_manifest": True,
                "pt_record_exists": pt_record is not None,
                "pt_record_valid": pt_valid if pt_record is not None else "",
                "pt_record_path": relative(pt_path) if pt_path else "",
                "pt_model_name_exact": bool(pt_record and pt_record.get("model_name") == model),
                "a_strict_success": a_pass,
                "b_strict_success": b_pass,
                "a_b_shared_strict_success": a_pass and b_pass,
                "pt_a_b_identity_matched": pt_valid and a_pass and b_pass,
                "pt_input_shape_nhwc": shape_string(pt_shape),
                "a_input_shape_nhwc": shape_string(a_shape),
                "a_input_shape_record_path": a_path or "",
                "pt_a_input_shape_match": shape_match if pt_shape and a_shape else "",
                "match_rule": "exact model_name equality; no aliases or fuzzy matching",
                "match_status": status,
            }
        )
        if not (pt_valid and a_pass and b_pass):
            continue
        assert pt_record is not None and pt_path is not None
        values = {
            "model_name": model,
            "pt_record_path": relative(pt_path),
            "pt_cpu_ms": float(pt_record["cpu_duration"]) / 1_000_000.0,
            "pt_gpu_ms": float(pt_record["gpu_duration"]) / 1_000_000.0,
            "a_cpu_ms": float(ab["a_cpu_mean_ms"]),
            "a_gpu_ms": float(ab["a_gpu_mean_ms"]),
            "b_cpu_ms": float(ab["b_cpu_mean_ms"]),
            "b_gpu_ms": float(ab["b_gpu_mean_ms"]),
            "pt_input_shape_nhwc": shape_string(pt_shape),
            "a_input_shape_nhwc": shape_string(a_shape),
            "pt_a_input_shape_match": shape_match,
        }
        for backend in ("cpu", "gpu"):
            for runtime in ("a", "b"):
                values[f"{runtime}_{backend}_minus_pt_ms"] = values[f"{runtime}_{backend}_ms"] - values[f"pt_{backend}_ms"]
                values[f"{runtime}_{backend}_over_pt"] = values[f"{runtime}_{backend}_ms"] / values[f"pt_{backend}_ms"]
        matched_rows.append(values)

    if not matched_rows:
        raise RuntimeError("No PT/A/B matched timing rows")
    write_csv(OUT / "model_overlap.csv", overlap_rows)
    write_csv(OUT / "matched_pt_a_b.csv", matched_rows)
    make_plot(matched_rows)

    a_pass = {row["model_name"] for row in ab_rows if as_bool(row["a_repeat_stable_pass"])}
    b_pass = {row["model_name"] for row in ab_rows if as_bool(row["b_repeat_stable_pass"])}
    pt_valid_models = {model for model, (_, record) in pt_catalog.items() if record.get("valid") is True}
    shape_aligned = [row for row in matched_rows if row["pt_a_input_shape_match"]]
    comparison = {
        "identity_matched": {
